### Single Universal Nano (NO subclasses — this is v2, not v1)

The 296 typed nano subclasses (FeatureNano, PatternNano, etc.) from v1 are DEAD. Do not reference them. A single Nano class handles all tasks. All specialization is emergent via learned RBY coordinates.

```python
class Nano(nn.Module):
    def __init__(self, d_model=256):
        super().__init__()
        self.fc1 = nn.Linear(d_model, d_model * 4)
        self.fc2 = nn.Linear(d_model * 4, d_model)
        self.norm = nn.LayerNorm(d_model)
        rby = torch.rand(3)
        self.rby = nn.Parameter(rby / rby.sum())  # simplex constraint
```

### ChromaticIndex (O(log N) routing via scipy KDTree)

```python
class ChromaticIndex:
    def _rebuild(self):
        coords = torch.stack([n.rby.detach() for n in self.nanos]).numpy()
        log_c = np.log(coords + 1e-8)
        centered = log_c - log_c.mean(axis=1, keepdims=True)  # Aitchison transform
        self.kdtree = KDTree(centered)  # scipy.spatial.KDTree

    def query(self, query_rby, k=50):
        q = query_rby.detach().numpy()
        log_q = np.log(q + 1e-8)
        c_q = log_q - log_q.mean()
        _, indices = self.kdtree.query(c_q, k=k)
        return indices.tolist()
```

### SoftKRouter (differentiable k via reverse cumsum — the key v2 innovation)

```python
class SoftKRouter(nn.Module):
    def __init__(self, d_model=256, n_candidates=50):
        super().__init__()
        self.score_proj = nn.Linear(d_model, n_candidates)
        self.k_threshold = nn.Parameter(torch.tensor(0.3))

    def forward(self, x, candidate_scores):
        probs = F.softmax(candidate_scores, dim=-1)
        sorted_probs, sort_idx = torch.sort(probs, dim=-1, descending=True)
        cumsum = torch.cumsum(sorted_probs, dim=-1)
        gates = (cumsum - sorted_probs < self.k_threshold.sigmoid()).float()
        w = sorted_probs * gates
        w = w / (w.sum(dim=-1, keepdim=True) + 1e-8)
        return w, sort_idx
```

### Training Loss (efficiency regularization prevents nano collapse)

```python
def compute_loss(logits, targets, router_weights):
    lm_loss = F.cross_entropy(logits.view(-1, VOCAB_SIZE), targets.view(-1))
    load = router_weights.mean(dim=[0, 1])
    eff_loss = EFF_LAMBDA * load.var()  # penalize uneven utilization
    return lm_loss + eff_loss
```
