class UnstoppableForce:
    def __init__(self, config: Dict):
        self.config = config
        self.base_urge = AEConstants.URGE_DEFAULT
        self.urge_history = []
        self.novelty_tracker = NoveltyTracker()
        
    def compute_urge(self, current_state: AEState, environment: Dict) -> float:
        """
        Compute unstoppable force based on:
        - Current understanding level (inverse relationship)
        - Environmental novelty
        - Internal homeostasis state
        - Historical patterns
        """
        # Base urge from understanding deficit
        understanding_deficit = 1.0 - current_state.understanding_level
        base_urge_component = self._sigmoid(understanding_deficit * 3.0)
        
        # Novelty component from environment
        environmental_novelty = self.novelty_tracker.compute_novelty(environment)
        novelty_component = environmental_novelty
        
        # Homeostasis modulation
        homeostasis = self._compute_rby_homeostasis(current_state.rby_balance)
        homeo_component = 1.0 if homeostasis >= AEConstants.MIN_HOMEOSTASIS else 0.3
        
        # Combine components
        raw_urge = (base_urge_component * 0.5 + 
                   novelty_component * 0.3 + 
                   self._temporal_urgency(current_state) * 0.2)
        
        # Apply homeostasis modulation
        modulated_urge = raw_urge * homeo_component
        
        # Bound urge
        final_urge = np.clip(modulated_urge * AEConstants.URGE_MAX, 
                           AEConstants.URGE_MIN, AEConstants.URGE_MAX)
        
        current_state.urge_level = final_urge
        self.urge_history.append(final_urge)
        
        return final_urge
    
    def _compute_rby_homeostasis(self, rby_balance: np.ndarray) -> float:
        """Compute RBY homeostasis metric"""
        deviation = np.sum(np.abs(rby_balance - AEConstants.RBY_TARGET))
        max_deviation = 2/3
        return 1.0 - (deviation / max_deviation)
    
    def _temporal_urgency(self, state: AEState) -> float:
        """Compute temporal urgency based on cycle patterns"""
        if len(state.touch_history) < 10:
            return 0.5
        
        # Compute success rate of recent touches
        recent_touches = state.touch_history[-10:]
        success_rate = np.mean([t['success'] for t in recent_touches])
        
        # Higher urgency when recent success is moderate (exploration sweet spot)
        return 2.0 * abs(success_rate - 0.5)  # Peaks at 0.5 success rate
    
    def _sigmoid(self, x: float) -> float:
        """Sigmoid function for smooth transitions"""
        return 1.0 / (1.0 + np.exp(-x))
    
    def modulate_policy_temperature(self, base_policy: np.ndarray, urge: float) -> np.ndarray:
        """
        Modulate policy exploration based on urge level
        Higher urge → higher temperature → more exploration
        """
        temperature = 1.0 + (urge - AEConstants.URGE_DEFAULT) * 0.5
        
        # Apply temperature scaling to policy
        if temperature == 1.0:
            return base_policy
        
        logits = np.log(base_policy + 1e-10)
        tempered_logits = logits / temperature
        tempered_policy = np.exp(tempered_logits)
        tempered_policy = tempered_policy / np.sum(tempered_policy)
        
        return tempered_policy

class UnderstandingFilter:

            'filtered_update': None

            return update_candidate

        agreement_check = self._check_ensemble_agreement(new_evidence)
        update_candidate['filter_results']['agreement'] = agreement_check

        consistency_check = self._check_consistency(current_state, new_evidence)
        update_candidate['filter_results']['consistency'] = consistency_check

            return update_candidate

        understanding_check = self._check_understanding_level(current_state, new_evidence)
        update_candidate['filter_results']['understanding'] = understanding_check

            # All filters passed - create filtered update
            filtered_update = self._create_filtered_update(current_state, new_evidence)
            update_candidate['passed_filters'] = True
            update_candidate['filtered_update'] = filtered_update

            self._update_understanding_level(current_state, new_evidence)

        # Understanding increases with successful integration of complex evidence
        complexity = evidence.get('complexity', 0.5)
        integration_success = evidence.get('integration_quality', 0.5)
        
        understanding_increase = 0.01 * complexity * integration_success
        current_state.understanding_level = min(
            1.0, current_state.understanding_level + understanding_increase
        )

class Crystallizer:
    def __init__(self, config: Dict):
        self.config = config
        self.compressor = MDLCompressor()
        self.artifact_generator = ArtifactGenerator()
        self.quality_assessor = QualityAssessor()
        
    def crystallize(self, current_state: AEState, 
                   touch_event: Dict, 
                   outcome: Dict) -> Dict:
        """
        Crystallize touch event into AEc artifact through:
        1. Minimum Description Length compression
        2. Quality assessment
        3. Artifact generation
        4. Integration into crystal store
        """
        # Step 1: Prepare crystallization candidate
        candidate = self._prepare_candidate(current_state, touch_event, outcome)
        
        # Step 2: Apply MDL compression
        compressed_artifact = self.compressor.compress(candidate)
        
        # Step 3: Assess crystallization quality
        quality_metrics = self.quality_assessor.assess(compressed_artifact)
        
        # Step 4: Generate final artifact if quality sufficient
        if quality_metrics['overall_quality'] >= AEConstants.MIN_CRYSTALLIZATION_QUALITY:
            ae_c_artifact = self.artifact_generator.generate(
                compressed_artifact, quality_metrics
            )
            
            # Step 5: Integrate into AEc store
            integration_result = current_state.ae_crystal.integrate(ae_c_artifact)
            
            crystallization_result = {
                'success': True,
                'artifact': ae_c_artifact,
                'quality_metrics': quality_metrics,
                'integration_result': integration_result,
                'compression_ratio': compressed_artifact['compression_ratio'],
                'information_gain': quality_metrics['information_gain']
            }
        else:
            crystallization_result = {
                'success': False,
                'reason': 'Insufficient quality',
                'quality_metrics': quality_metrics
            }
        
        return crystallization_result
    
    def _prepare_candidate(self, state: AEState, touch_event: Dict, outcome: Dict) -> Dict:
        """Prepare candidate for crystallization"""
        return {
            'internal_state_snapshot': state.internal_state,
            'touch_event': touch_event,
            'outcome': outcome,
            'pre_touch_understanding': state.understanding_level,
            'rby_balance': state.rby_balance.copy(),
            'urge_level': state.urge_level,
            'timestamp': time.time(),
            'cycle_count': state.cycle_count
        }
    
    def compute_crystallization_yield(self, crystallization_result: Dict) -> float:
        """Compute crystallization yield (information gain per touch)"""
        if not crystallization_result['success']:
            return 0.0
        
        info_gain = crystallization_result['information_gain']
        compression_ratio = crystallization_result['compression_ratio']
        
        # Yield combines information gain and compression efficiency
        base_yield = info_gain * compression_ratio
        
        # Quality bonus for high-quality crystallizations
        quality = crystallization_result['quality_metrics']['overall_quality']
        quality_bonus = 1.0 + (quality - 0.5)  # 0.5-1.5 range
        
        return base_yield * quality_bonus

## 5. AEC Crystal Store

```python
class AECrystal:
    def __init__(self):
        self.artifacts = []  # Crystallized knowledge artifacts
        self.index = KnowledgeIndex()
        self.compression_stats = {
            'total_information_gain': 0.0,
            'average_compression_ratio': 0.0,
            'artifact_count': 0
        }
    
    def integrate(self, artifact: Dict) -> Dict:
        """Integrate new artifact into crystal store"""
        # Check for duplicates/similar artifacts
        similarity_scores = [
            self._compute_similarity(artifact, existing) 
            for existing in self.artifacts
        ]
        
        if similarity_scores and max(similarity_scores) > 0.9:
            # High similarity - merge instead of adding new
            return self._merge_artifact(artifact, similarity_scores)
        else:
            # New artifact - add to store
            self.artifacts.append(artifact)
            self._update_index(artifact)
            self._update_stats(artifact)
            
            return {
                'integration_type': 'new',
                'artifact_id': len(self.artifacts) - 1,
                'information_gain': artifact.get('information_gain', 0.0)
            }
    
    def _compute_similarity(self, artifact1: Dict, artifact2: Dict) -> float:
        """Compute similarity between two artifacts"""
        # Compare embeddings, content hashes, and semantic meaning
        embedding_sim = self._embedding_similarity(
            artifact1.get('embedding'), artifact2.get('embedding')
        )
        
        content_sim = self._content_similarity(
            artifact1.get('content'), artifact2.get('content')
        )
        
        return 0.6 * embedding_sim + 0.4 * content_sim
    
    def _merge_artifact(self, new_artifact: Dict, similarity_scores: List[float]) -> Dict:
        """Merge new artifact with most similar existing artifact"""
        most_similar_idx = np.argmax(similarity_scores)
        existing_artifact = self.artifacts[most_similar_idx]
        
        # Merge content and update information gain
        merged_artifact = self._merge_content(existing_artifact, new_artifact)
        
        # Replace existing artifact with merged version
        self.artifacts[most_similar_idx] = merged_artifact
        self._update_index(merged_artifact, update=True)
        
        return {
            'integration_type': 'merge',
            'merged_with': most_similar_idx,
            'similarity': similarity_scores[most_similar_idx],
            'information_gain': merged_artifact.get('information_gain', 0.0)
        }
    
    def query(self, query_embedding: torch.Tensor, max_results: int = 5) -> List[Dict]:
        """Query crystal store for relevant knowledge"""
        similarities = []
        for i, artifact in enumerate(self.artifacts):
            sim = self._embedding_similarity(
                query_embedding, artifact.get('embedding')
            )
            similarities.append((i, sim, artifact))
        
        # Sort by similarity and return top results
        similarities.sort(key=lambda x: x[1], reverse=True)
        return [item[2] for item in similarities[:max_results]]
    
    def get_crystallization_volume(self) -> float:
        """Compute current AEc volume metric"""
        if not self.artifacts:
            return 0.0
        
        # Volume components (normalized)
        artifact_count = len(self.artifacts) / 100.0  # Normalized
        total_information = self.compression_stats['total_information_gain']
        avg_compression = self.compression_stats['average_compression_ratio']
        
        # Index quality (simplified)
        index_quality = self.index.get_quality()
        
        # Composite volume metric
        volume = (0.3 * artifact_count + 
                 0.4 * total_information + 
                 0.2 * avg_compression + 
                 0.1 * index_quality)
        
        return volume
```

## 6. Main AE(IO+UF)→AEc Engine

```python
class AECrystallizationEngine:
    def __init__(self, config: Dict):
        self.config = config
        self.state = AEState()
        
        # Core modules
        self.urge_force = UrgeForce(config)
        self.immovable_object = ImmovableObject(config)
        self.understanding_filter = UnderstandingFilter(config)
        self.crystallizer = Crystallizer(config)
        self.touch_executor = TouchExecutor(config)
        
        # Monitoring
        self.metrics_tracker = CrystallizationMetricsTracker()
        self.absularity_detector = AbsularityDetector()
        
    def execute_cycle(self, environment: Dict) -> Dict:
        """
        Execute complete AE(IO+UF)→AEc cycle:
        1. Compute unstoppable force
        2. Generate imaginative touch plans
        3. Execute touch
        4. Filter understanding
        5. Crystallize results
        """
        cycle_result = {
            'cycle_number': self.state.cycle_count,
            'success': False,
            'crystallization_yield': 0.0,
            'understanding_change': 0.0,
            'urge_level': 0.0
        }
        
        # Step 1: Compute Unstoppable Force
        urge = self.urge_force.compute_urge(self.state, environment)
        cycle_result['urge_level'] = urge
        
        # Step 2: Immovable Object generates touch plans
        touch_plans = self.immovable_object.generate_touch_plans(
            self.state, environment, urge
        )
        
        if not touch_plans:
            cycle_result['error'] = 'No valid touch plans generated'
            return cycle_result
        
        # Step 3: Select and execute touch
        selected_plan = touch_plans[0]  # Highest ranked plan
        touch_result = self.touch_executor.execute(
            selected_plan['action'], environment
        )
        
        # Step 4: Process outcome through understanding filter
        filter_result = self.understanding_filter.filter_update(
            self.state, touch_result, urge
        )
        
        if filter_result['passed_filters']:
            # Step 5: Crystallize successful understanding
            crystallization_result = self.crystallizer.crystallize(
                self.state, selected_plan, touch_result
            )
            
            if crystallization_result['success']:
                # Update internal state with filtered understanding
                self._update_internal_state(filter_result['filtered_update'])
                
                # Update metrics
                cycle_result['crystallization_yield'] = (
                    self.crystallizer.compute_crystallization_yield(crystallization_result)
                )
                cycle_result['understanding_change'] = (
                    self.state.understanding_level - cycle_result.get('initial_understanding', 0.0)
                )
                cycle_result['success'] = True
                
                # Record touch history
                self.state.touch_history.append({
                    'plan': selected_plan,
                    'result': touch_result,
                    'crystallization': crystallization_result,
                    'urge_level': urge,
                    'cycle': self.state.cycle_count
                })
        
        # Step 6: Check for absularity (cycle completion point)
        absularity_detected = self.absularity_detector.check(
            self.state, cycle_result
        )
        
        if absularity_detected:
            cycle_result['absularity_detected'] = True
            cycle_result['absularis_snapshot'] = self._capture_absularis_snapshot()
        
        self.state.cycle_count += 1
        return cycle_result
    
    def _update_internal_state(self, filtered_update: Dict):
        """Update internal state with filtered understanding"""
        integration_weight = filtered_update.get('integration_weight', 0.1)
        
        # Gradual integration of new understanding
        if self.state.internal_state is None:
            self.state.internal_state = filtered_update['content']
        else:
            # Weighted update based on understanding level
            self.state.internal_state = (
                (1 - integration_weight) * self.state.internal_state +
                integration_weight * filtered_update['content']
            )
    
    def _capture_absularis_snapshot(self) -> Dict:
        """Capture absularis snapshot at cycle completion point"""
        snapshot = {
            'timestamp': time.time(),
            'cycle_count': self.state.cycle_count,
            'internal_state': self.state.internal_state.copy() if self.state.internal_state is not None else None,
            'ae_crystal_state': self.state.ae_crystal.get_state(),
            'understanding_level': self.state.understanding_level,
            'urge_level': self.state.urge_level,
            'rby_balance': self.state.rby_balance.copy(),
            'touch_history_summary': self._summarize_touch_history(),
            'merkle_root': self._compute_merkle_root()
        }
        
        return snapshot
    
    def _compute_merkle_root(self) -> str:
        """Compute Merkle root for reproducibility"""
        content = f"{self.state.cycle_count}:{self.state.understanding_level}:{self.state.urge_level}"
        return hashlib.sha256(content.encode()).hexdigest()
```

        results = {}
        for condition_name, condition_config in conditions:
            print(f"Running condition: {condition_name}")

            results[condition_name] = condition_results
        
        return results
    
    def run_urge_sweep(self, urge_levels: List[float], cycles_per_level: int = 100) -> Dict:
        """Run urge sweep experiment for H2 testing"""
        results = {}
        
        for urge_level in urge_levels:
            config = {
                'fixed_urge_level': urge_level,
                'use_io': True,
                'use_uf': False,  # Use fixed urge instead of dynamic UF
                'use_understanding_filter': True
            }
            
            level_results = self._run_condition(config, cycles_per_level)
            results[urge_level] = {
                'understanding': np.mean(level_results['understanding_level']),
                'novelty': np.mean(level_results['novelty_scores']),
                'destruction_ratio': np.mean(level_results['destruction_ratio']),
                'crystallization_yield': np.mean(level_results['crystallization_yield'])
            }
        
        return results
    
    def run_multi_cycle_meta(self, num_cycles: int = 5, cycles_per_meta: int = 200) -> Dict:
        """Run multi-cycle meta-experiment for H4 testing"""
        meta_results = []
        
        for cycle in range(num_cycles):
            print(f"Meta-cycle {cycle + 1}/{num_cycles}")
            
            # Start from absularis snapshot if available
            if cycle > 0 and meta_results[cycle-1].get('absularis_snapshot'):
                initial_state = meta_results[cycle-1]['absularis_snapshot']
            else:
                initial_state = None
            
            cycle_results = self._run_meta_cycle(cycles_per_meta, initial_state)
            meta_results.append(cycle_results)
        
        # Analyze meta-learning trend
        initial_understandings = [
            r['initial_understanding'] for r in meta_results
        ]
        final_understandings = [
            r['final_understanding'] for r in meta_results
        ]
        
        return {
            'meta_cycles': meta_results,
            'initial_understanding_trend': initial_understandings,
            'final_understanding_trend': final_understandings,
            'improvement_trend': self._compute_improvement_trend(initial_understandings)
        }

        """Run a single experimental condition"""
        engine = AECrystallizationEngine(config)
        results = {
            'understanding_level': [],
            'crystallization_yield': [],
            'novelty_scores': [],
            'destruction_ratio': [],
            'urge_levels': []
        }
        
        destruction_events = 0
        
        for cycle in range(num_cycles):
            environment = self._generate_environment(cycle)
            cycle_result = engine.execute_cycle(environment)
            
            results['understanding_level'].append(engine.state.understanding_level)
            results['crystallization_yield'].append(cycle_result.get('crystallization_yield', 0.0))
            results['urge_levels'].append(cycle_result.get('urge_level', 0.0))
            
            # Track destruction events
            if cycle_result.get('understanding_change', 0) < -0.1:  # Significant understanding drop
                destruction_events += 1
            
            # Track novelty
            if cycle_result.get('touch_result'):
                results['novelty_scores'].append(
                    cycle_result['touch_result'].get('novelty', 0.0)
                )
        
        results['destruction_ratio'] = destruction_events / num_cycles
        return results
