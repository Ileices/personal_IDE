        fractal_state.current_fic = fic
        return fic
    
    def _compute_scale_bandwidth(self, state: FractalState) -> float:
        """Compute scale bandwidth B_s = log2(s_max/s_min)"""
        scales = list(state.scale_embeddings.keys())
        if len(scales) < 2:
            return 0.0
        max_scale = max(scales)
        min_scale = min(scales)
        return np.log2(max_scale / min_scale) / SystemConstants.FRACTAL_OCTAVES
    
    def _compute_cross_scale_similarity(self, state: FractalState) -> float:
        """Compute normalized mutual information between adjacent scales"""
        if len(state.cross_scale_similarities) == 0:
            return 0.0
        return np.mean(state.cross_scale_similarities)
    
    def _compute_compression_gain(self, state: FractalState) -> float:
        """Compute minimum description length compression gain"""
        if len(state.compression_gains) == 0:
            return 0.0
        baseline_entropy = 8.0  # Assume 8 bits per pixel baseline
        current_entropy = baseline_entropy * (1 - np.mean(state.compression_gains))
        return (baseline_entropy - current_entropy) / baseline_entropy
    
    def _compute_weight_self_similarity(self, state: FractalState) -> float:
        """Compute cosine similarity between weights at different scales"""
        if len(state.weight_self_similarities) == 0:
            return 0.0
        return np.mean(state.weight_self_similarities)

        x_flat = x.flatten().detach().cpu().numpy()
        y_flat = y.flatten().detach().cpu().numpy()

class MultiScaleProcessor:

            # Process at each scale
            scale_embedding = self._process_single_scale(scale_data, scale_factor)
            fractal_state.scale_embeddings[scale_factor] = scale_embedding
        
        # Compute cross-scale relationships
        self._compute_cross_scale_relationships(fractal_state)
        
        # Compute FIC
        self.fic_calculator.compute_fic(fractal_state)

    def _process_single_scale(self, scale_data: torch.Tensor, scale_factor: float) -> torch.Tensor:
        """Process data at a single scale"""
        scale_processor = self.scale_processors[scale_factor]
        return scale_processor(scale_data)
    
    def _compute_cross_scale_relationships(self, state: FractalState):
        """Compute relationships between adjacent scales"""
        scales = sorted(state.scale_embeddings.keys())
        
        for i in range(len(scales) - 1):
            current_scale = scales[i]
            next_scale = scales[i + 1]
            
            current_embedding = state.scale_embeddings[current_scale]
            next_embedding = state.scale_embeddings[next_scale]
            
            # Downsample next scale to match current scale
            next_downsampled = nn.functional.interpolate(
                next_embedding, size=current_embedding.shape[2:]
            )
            
            # Compute similarity
            similarity = self._compute_embedding_similarity(current_embedding, next_downsampled)
            state.cross_scale_similarities.append(similarity)
