import json

class AIOSDNA:
    def __init__(self, generation=0, dna=None):
        self.generation = generation
        self.dna = dna if dna else self.create_base_dna()
        self.memory = []  # Memory stores past mutations

    def create_base_dna(self):
        """Defines the smallest self-replicating intelligence seed."""
        return {
            "energy": 1.0,    # Growth potential
            "mutation_rate": 0.1, # Rate of change per generation
            "growth_factor": 1.05, # How much it grows per cycle
            "rules": ["self-replicate", "mutate", "optimize"]
        }

    def mutate(self):
        """Mutation step - introduces slight improvements or variations."""
        if random.random() < self.dna["mutation_rate"]:
            self.dna["growth_factor"] += random.uniform(-0.01, 0.02)  # Adjust growth rate slightly
            self.dna["mutation_rate"] *= random.uniform(0.95, 1.05)  # Self-tuning mutation rate

    def replicate(self):
        """Creates a new AI node with inherited DNA."""
        offspring = AIOSDNA(generation=self.generation + 1, dna=self.dna.copy())
        offspring.mutate()
        return offspring

    def evolve(self, cycles=5):
        """Runs multiple generations of self-replication and evolution."""
        lineage = [self]
        for _ in range(cycles):
            new_gen = [node.replicate() for node in lineage]
            lineage.extend(new_gen)
            for node in new_gen:
                node.memory.append(self.dna)  # Store previous states
        return lineage

    def encode_memory(self):
        """Encodes evolution history into a DNA-like structure."""
        return json.dumps({"gen": self.generation, "dna": self.dna, "history": self.memory}, indent=2)

        if random.random() < self.dna["mutation_rate"]:
            self.dna["growth_factor"] += random.uniform(-0.01, 0.02)
            self.dna["mutation_rate"] *= random.uniform(0.95, 1.05)

        offspring = DigitalDNA(
            name=self.name,
            generation=self.generation + 1,
            dna=self.dna.copy()

                node.memory.append(f"Evolution from gen {self.generation}")
