# guardrails.py
from __future__ import annotations
from pathlib import Path
from typing import Iterable
import os

class FileGate:
    """
    Enforces read-only access outside an organism's workspace.
    All writes must live under `workspace_root`.
    """
    def __init__(self, workspace_root: Path):
        self.workspace_root = workspace_root.resolve()

    def assert_write_path(self, p: Path) -> Path:
        p = p.resolve()
        if not str(p).startswith(str(self.workspace_root)):
            raise PermissionError(f"Write denied outside workspace: {p}")
        p.parent.mkdir(parents=True, exist_ok=True)
        return p

    def read_text(self, p: Path, encoding="utf-8") -> str:
        return Path(p).read_text(encoding=encoding, errors="ignore")

    def write_text(self, p: Path, data: str, encoding="utf-8") -> None:
        p = self.assert_write_path(p)
        p.write_text(data, encoding=encoding)

    def write_bytes(self, p: Path, data: bytes) -> None:
        p = self.assert_write_path(p)
        p.write_bytes(data)

    def copy_into_workspace(self, src: Path, dst_rel: Path) -> Path:
        from shutil import copy2
        dst = self.assert_write_path(self.workspace_root / dst_rel)
        copy2(src, dst)
        return dst

def safe_glob(root: Path, patterns: Iterable[str]):
    for pat in patterns:
        yield from root.glob(pat)

# indexer.py
from pathlib import Path
from typing import List, Tuple
from settings import SAFE_READ_GLOBS, HIGH_VALUE_EXT
from guardrails import safe_glob

def recursive_index(root: Path) -> List[Path]:
    items = []
    for p in safe_glob(root, SAFE_READ_GLOBS):
        try:
            if p.is_file():
                items.append(p)
        except Exception:
            continue
    return items

def score_files(paths: List[Path]) -> List[Tuple[Path, float]]:
    scored = []
    for p in paths:
        ext = p.suffix.lower()
        w = 1.0
        if ext in HIGH_VALUE_EXT: w += 2.0
        if "config" in p.name.lower(): w += 0.5
        if "log" in p.name.lower(): w += 0.3
        if "readme" in p.name.lower(): w += 0.2
        scored.append((p, w))
    scored.sort(key=lambda t: t[1], reverse=True)
    return scored

try:
    import psutil
except ImportError:
    psutil = None

class ResourceSnapshot(NamedTuple):
    disk_used_ratio: float
    cpu_util: float
    ram_util: float

def snapshot() -> ResourceSnapshot:
    if psutil is None:
        # fallbacks
        import shutil, os
        total, used, free = shutil.disk_usage("/")
        disk = used / total if total else 0.0
        return ResourceSnapshot(disk, 0.2, 0.2)
    disk = psutil.disk_usage("/").percent / 100.0
    cpu = psutil.cpu_percent(interval=0.2) / 100.0
    ram = psutil.virtual_memory().percent / 100.0
    return ResourceSnapshot(disk, cpu, ram)

def overloaded(disk_thr, cpu_thr, ram_thr) -> bool:
    s = snapshot()
    return (s.disk_used_ratio >= disk_thr) or (s.cpu_util >= cpu_thr) or (s.ram_util >= ram_thr)

from __future__ import annotations
from dataclasses import dataclass, asdict
from typing import Dict, Any
import random, math, json

@dataclass
class Genome:
    """
    Minimal, extensible genome.
    """
    # perception / learning weights
    global_bias: float = 0.5
    local_bias: float = 0.5
    prioritise_logs: float = 0.8
    prioritise_scripts: float = 0.9
    prioritise_configs: float = 0.7
    multitask_level: float = 0.1
    gpu_affinity: float = 0.05
    compression_preference: float = 0.4
    error_tolerance: float = 0.2

    # phase marker: 0 bacteria, 1 animal threshold drive
    complexity_drive: float = 0.2

    def to_json(self) -> str:
        return json.dumps(asdict(self), separators=(",", ":"))

    @staticmethod
    def from_json(s: str) -> Genome:
        d = json.loads(s)
        return Genome(**d)

    def intelligence_score(self) -> float:
        v = (self.multitask_level + self.gpu_affinity + self.compression_preference*0.5 +
             self.global_bias*0.25 + self.local_bias*0.25 + (1.0 - self.error_tolerance)*0.2)
        return max(0.0, min(1.0, v))

    def mutate(self, rate: float) -> Genome:
        d = asdict(self)
        for k, v in d.items():
            if isinstance(v, float) and random.random() < rate:
                delta = random.uniform(-0.15, 0.15)
                d[k] = max(0.0, min(1.0, v + delta))
        return Genome(**d)

def crossover(a: Genome, b: Genome, rate: float) -> Genome:
    da, db = asdict(a), asdict(b)
    out = {}
    for k in da:
        if isinstance(da[k], float):
            if random.random() < rate:
                # arithmetic crossover
                w = random.random()
                out[k] = max(0.0, min(1.0, w*da[k] + (1-w)*db[k]))
            else:
                out[k] = random.choice([da[k], db[k]])
        else:
            out[k] = da[k]
    return Genome(**out)

        g_weight = self.genome.global_bias

    def maybe_phase_shift(self):
        iq = self.genome.intelligence_score()
        if iq >= BACTERIA_IQ_THRESHOLD and self.phase == "bacterial":
            self.phase = "animal"
            log_event(self.gate, "phase_shift", {"to": self.phase, "iq": iq})

    def reproduce(self, partner: Optional["Organism"]=None) -> Optional["Organism"]:
        if self.phase == "bacterial":
            if self.genome.intelligence_score() < INTELLIGENCE_TARGET: return None
            child_genome = self.genome.mutate(MUTATION_RATE)
            child = Organism._spawn_child(self, None, child_genome)
            return child
        # animal phase: require partner and diversity
        if partner is None: return None
        if partner.phase != "animal": return None

        # diversity check via L2 distance on genes
        import numpy as np
        a = np.array(list(compact_genes(self.genome.__dict__).values()), dtype=float)
        b = np.array(list(compact_genes(partner.genome.__dict__).values()), dtype=float)
        if a.size != b.size: return None
        # cosine distance
        sim = float(np.dot(a,b) / (np.linalg.norm(a)*np.linalg.norm(b) + 1e-9))
        if 1.0 - sim < 0.2:  # insufficient diversity
            return None
        mixed = crossover(self.genome, partner.genome, CROSSOVER_RATE).mutate(MUTATION_RATE*0.5)
        child = Organism._spawn_child(self, partner, mixed)
        return child

    @staticmethod
    def _spawn_child(a: "Organism", b: Optional["Organism"], genome: Genome) -> "Organism":
        # select a fresh environment for the child (isolation)
        all_roots = [DATA_POOL_DIR.parent, a.env_selected.parent, a.workspace.parent]
        # choose best-scoring dir by contents
        candidates = []
        for r in all_roots:
            if r.exists():
                items = recursive_index(r)
                candidates.append((r, len(items)))
        candidates.sort(key=lambda t: t[1], reverse=True)
        env = candidates[0][0] if candidates else a.env_selected

        child_id = uuid.uuid4().hex[:16]
        ws = SPLIT_ORGS_DIR / f"{int(time.time())}_{child_id}"
        ws.mkdir(parents=True, exist_ok=True)

        child = Organism(child_id, genome, a.generation+1, env.resolve(), ws.resolve(), "bacterial", FileGate(ws), a.pool)
        from registry import OrgRow, upsert_org
        row = OrgRow(
            id=child_id, parent_a=a.id, parent_b=(b.id if b else None),
            generation=child.generation, birth_ts=time.time(), death_ts=None,
            env_selected=str(env), workspace=str(ws), genome_json=genome.to_json(),
            intelligence=genome.intelligence_score(), phase=child.phase
        )
        upsert_org(row)
        # record heredity
        log_event(child.gate, "born", {"parent_a": a.id, "parent_b": (b.id if b else None), "genome": compact_genes(genome.__dict__)})
        return child

def cull_policy(org_rows) -> List[str]:
    """
    Kill weakest first; if system overloaded or population beyond MAX_POP.
    """
    alive = list(org_rows)
    if not alive: return []
    # sort by intelligence ascending, older first
    alive.sort(key=lambda r: (r["intelligence"], -r["generation"]))
    to_kill = []
    # enforce pop cap
    if len(alive) > MAX_POP:
        to_kill.extend([r["id"] for r in alive[:len(alive)-MAX_POP]])

    if overloaded(DISK_DANGER_THRESHOLD, CPU_OVERLOAD_THRESHOLD, RAM_OVERLOAD_THRESHOLD):
        # kill 10% weakest
        n = max(1, int(0.1 * len(alive)))
        to_kill.extend([r["id"] for r in alive[:n]])

    return list(dict.fromkeys(to_kill))  # unique, preserve order

def reap(org_ids: List[str]):
    now = time.time()
    for oid in org_ids:
        mark_dead(oid, now)

def mutate_text(source: str, power: float) -> str:
    """
    Non-destructive mutation: small refactors, doc additions, config smoothing.
    """
    lines = source.splitlines()
    out = []
    for ln in lines:
        if ln.strip().startswith("#") and random.random() < power*0.3:
            out.append(ln + "  # refined")
        elif "TODO" in ln and random.random() < power*0.5:
            out.append(ln.replace("TODO", "DONE"))
        else:
            out.append(ln)
    if random.random() < power*0.2 and len(out) > 5:
        out.insert(0, "# auto-annotated by organism\n")
    return "\n".join(out)

def mutate_file(gate: FileGate, src: Path, rel_dst: Path, power: float=0.2):
    try:
        text = gate.read_text(src)
    except Exception:
        return
    mutated = mutate_text(text, power)
    gate.write_text(gate.workspace_root / rel_dst, mutated)

def apply_mutations(gate: FileGate, sources: list[Path], local_bias: float):

    Copy and mutate prioritized sources into the workspace.

        rel = Path("workspace_mosaic") / src.name

SCRIPT_PATTERNS = [
    "*.py","*.js","*.ts","*.mjs","*.c","*.cc","*.cpp","*.h","*.hpp","*.java","*.cs","*.rs","*.go",
    "*.sh","*.bash","*.ps1","*.yaml","*.yml","*.toml","*.ini","*.cfg","*.json","*.sql","*.ipynb",
]
LOG_PATTERNS = ["*.log","*.out","*.err","*.trace"]
CONFIG_PATTERNS = ["*.env",".env*","*.conf","*.config","Dockerfile","docker-compose*.yml"]

EXCLUDE_DIRS = {".git","node_modules","venv",".venv","__pycache__",".idea",".vscode"}
MAX_INDEX_FILES_PER_ROOT = 10000

@dataclasses.dataclass
class IndexedFile:
    path: Path
    size: int
    mtime: float
    score: float
    kind: str  # script/config/log/other

class FileIndexer:
    def __init__(self, roots: List[Path]):
        self.roots = roots
        self.cache: Dict[str, IndexedFile] = {}

    @staticmethod
    def _kind_for(name: str) -> str:
        for pat in SCRIPT_PATTERNS:
            if fnmatch.fnmatch(name, pat):
                return "script"
        for pat in CONFIG_PATTERNS:
            if fnmatch.fnmatch(name, pat):
                return "config"
        for pat in LOG_PATTERNS:
            if fnmatch.fnmatch(name, pat):
                return "log"
        return "other"

    @staticmethod
    def _score(kind: str, size: int, mtime: float) -> float:
        # Prioritize scripts > configs > logs > others; prefer smaller files and recent mtimes
        base = {"script": 10.0, "config": 7.0, "log": 5.0, "other": 1.0}.get(kind, 1.0)
        size_penalty = 1.0 / (1.0 + math.log2(1 + size))
        age_bonus = 1.0 / (1.0 + math.exp(-(time.time() - mtime)/86400.0))  # recent→~0.5..1.0
        return base * size_penalty * age_bonus

    def index(self) -> List[IndexedFile]:
        results: List[IndexedFile] = []
        seen = 0
        for root in self.roots:
            try:
                for dirpath, dirnames, filenames in os.walk(root, topdown=True):
                    # prune excluded dirs
                    dirnames[:] = [d for d in dirnames if d not in EXCLUDE_DIRS]
                    for name in filenames:
                        try:
                            path = Path(dirpath) / name
                            stat = path.stat()
                            kind = self._kind_for(name)
                            sc = self._score(kind, stat.st_size, stat.st_mtime)
                            results.append(IndexedFile(path=path, size=stat.st_size, mtime=stat.st_mtime, score=sc, kind=kind))
                            seen += 1
                            if seen >= MAX_INDEX_FILES_PER_ROOT:
                                break
                        except Exception:
                            continue
                    if seen >= MAX_INDEX_FILES_PER_ROOT:
                        break
            except Exception as e:
                Log.warn(f"Index error on {root}: {e}")
        results.sort(key=lambda f: f.score, reverse=True)
        Log.info(f"Indexed {len(results)} files across {len(self.roots)} roots")
        return results

class DataPool:

        self.indexer = FileIndexer([root])

        if not self._cache:

        return self._cache[:k]

@dataclasses.dataclass
class DNA:
    generation: int = 0
    lineage: List[str] = dataclasses.field(default_factory=list)  # parent ids
    traits: Dict[str, float] = dataclasses.field(default_factory=lambda: {
        "curiosity": 0.5,
        "risk": 0.5,
        "efficiency": 0.5,
        "parallelism": 0.2,
        "gpu_affinity": 0.1,
        "defense": 0.1,
        "compression": 0.4,
        "pattern_recognition": 0.3,
    })
    tech_tree: Dict[str, int] = dataclasses.field(default_factory=lambda: defaultdict(int))  # discoveries and levels

    def mutate(self, seed: int) -> 'DNA':
        rnd = random.Random(seed)
        new_traits = dict(self.traits)
        for k in new_traits:
            delta = rnd.uniform(-0.08, 0.08)
            new_traits[k] = float(min(1.0, max(0.0, new_traits[k] + delta)))
        new_tree = dict(self.tech_tree)
        # Rare breakthroughs
        if rnd.random() < 0.03:
            breakthrough = rnd.choice(["gpu", "parallel", "viz", "distributed", "auto_exec", "nlp"])
            new_tree[breakthrough] = new_tree.get(breakthrough, 0) + 1
        return DNA(generation=self.generation+1, lineage=list(self.lineage), traits=new_traits, tech_tree=new_tree)

    def mix(self, other: 'DNA', seed: int) -> 'DNA':
        rnd = random.Random(seed)
        mixed_traits = {k: (self.traits.get(k,0.5)+other.traits.get(k,0.5))/2 for k in set(self.traits)|set(other.traits)}
        # small cross-over noise
        for k in mixed_traits:
            mixed_traits[k] = float(min(1.0, max(0.0, mixed_traits[k] + rnd.uniform(-0.05, 0.05))))
        mixed_tree = dict(self.tech_tree)
        for k,v in other.tech_tree.items():
            mixed_tree[k] = max(mixed_tree.get(k,0), v)
        return DNA(generation=max(self.generation, other.generation)+1, lineage=list(set(self.lineage+other.lineage)), traits=mixed_traits, tech_tree=mixed_tree)

ELE_TEMPLATES = [
    {"type":"flood","effect":"data_corruption","severity":0.4},
    {"type":"fire","effect":"cpu_overload","severity":0.6},
    {"type":"freeze","effect":"io_lock","severity":0.5},
    {"type":"meteor","effect":"env_reset","severity":0.7},
    {"type":"pandemic","effect":"code_infection","severity":0.5},
]

def random_ele(seed: int) -> Dict[str, Any]:
    rnd = random.Random(seed)
    event = dict(rnd.choice(ELE_TEMPLATES))
    event["timestamp"] = now()
    event["severity"] = float(min(0.95, max(0.05, event["severity"] + rnd.uniform(-0.1, 0.1))))
    event["id"] = hashlib.sha1(f"{event['type']}-{event['timestamp']}-{seed}".encode()).hexdigest()[:10]
    return event

class Organism:
    def __init__(self, oid: str, env_path: Path, dna: DNA, data_pool: DataPool):
        self.id = oid
        self.env_path = env_path.resolve()
        self.dna = dna
        self.data_pool = data_pool
        self.birth = time.time()
        self.workspace = SPLIT_ROOT / oid
        self.workspace.mkdir(parents=True, exist_ok=True)
        self.log_path = LOG_ROOT / f"organism_{oid}.jsonl"
        self.alive = True
        self.energy = 1.0  # abstract survival energy
        self.fitness = 0.0
        self.phase = "bacterial"  # or "animal"
        self._indexer = FileIndexer([self.env_path, self.workspace])
        self._local_cache: List[IndexedFile] = []

    # ---------- Logging ----------
    def _log(self, record: Dict[str, Any]):
        record["ts"] = now()
        record["organism_id"] = self.id
        SafeWriter.write_text(self.log_path, json.dumps(record, ensure_ascii=False) + "\n")

    # ---------- Learning ----------
    def refresh_local_index(self):
        self._local_cache = self._indexer.index()

    def learn(self):
        # Prioritize scripts/configs/logs: combine data pool + local environment
        local_top = self._local_cache[:50]
        global_top = self.data_pool.topk(50)
        sample = local_top[:30] + global_top[:20]
        # simple content sampling
        absorbed = 0
        for f in sample[:10]:
            try:
                if f.size > 512_000:  # 500KB cap
                    continue
                text = Path(f.path).read_text(errors="ignore")
                digest = hashlib.sha256(text.encode("utf-8","ignore")).hexdigest()[:12]
                self._log({"event":"learn","file":str(f.path),"digest":digest,"kind":f.kind,"score":f.score})
                absorbed += 1
            except Exception as e:
                continue
        # Increase fitness by absorbed knowledge and traits
        self.fitness += absorbed * (0.05 + 0.1*self.dna.traits.get("pattern_recognition",0.3))
        # Phase transition trigger
        if self.dna.tech_tree.get("gpu",0) + self.dna.tech_tree.get("parallel",0) + self.dna.tech_tree.get("nlp",0) >= 2:
            self.phase = "animal"
        self._log({"event":"phase","phase":self.phase,"fitness":self.fitness})

    # ---------- Mutation ----------
    def mutate(self):
        seed = int.from_bytes(os.urandom(4), 'little')
        mode = "rewrite" if random.random() < self.dna.traits.get("risk",0.5) else "augment"
        # Choose a target intelligently from local index preferring scripts/configs
        candidates = [f for f in self._local_cache if f.kind in ("script","config","log")]
        target: Optional[IndexedFile] = None
        if candidates:
            target = max(candidates, key=lambda f: f.score * (1.0+random.random()*0.1))
        else:
            targets = self._local_cache[:10]
            target = targets[0] if targets else None
        result = {"event":"mutate","mode":mode}
        try:
            if target and target.size < 512_000:
                content = Path(target.path).read_text(errors="ignore")
                if mode == "augment":
                    patch = f"\n\n# AIOSIO[{self.id}] note:{seed} fitness={self.fitness:.3f}\n"
                    new_content = content + patch
                else:
                    new_content = self._simple_rewrite(content, seed)
                # write only inside workspace, mirrored path
                rel = self._mirror_relative(target.path)
                out_path = self.workspace / rel
                SafeWriter.write_text(out_path, new_content)
                result.update({"target":str(target.path),"output":str(out_path)})
                self.fitness += 0.02
            else:
                # create a new artifact from scratch
                out_path = self.workspace / f"artifact_{seed % 1_000_000}.txt"
                blob = f"artifact by {self.id} seed={seed} fitness={self.fitness:.3f}\n"
                SafeWriter.write_text(out_path, blob)
                result.update({"target":None,"output":str(out_path)})
                self.fitness += 0.01
        except Exception as e:
            result.update({"error":str(e)})
        self._log(result)

    def _simple_rewrite(self, content: str, seed: int) -> str:
        rnd = random.Random(seed)
        lines = content.splitlines()
        # replace some constants and identifiers heuristically
        for i in range(0, len(lines), max(1, len(lines)//50 or 1)):
            if i < len(lines):
                lines[i] = re.sub(r"(TODO|FIXME)", "DONE", lines[i])
                lines[i] = re.sub(r"(temp|tmp)", f"aiosio_{rnd.randint(0,999)}", lines[i])
        return "\n".join(lines)

    def _mirror_relative(self, path: Path) -> Path:
        # compute a stable relative subpath for mirrored writes into workspace
        try:
            rel = path.relative_to(self.env_path)
        except ValueError:
            try:
                rel = path.relative_to(ROOT)
            except ValueError:
                rel = Path("ext") / path.name
        return rel

    # ---------- Reproduction ----------
    def should_reproduce(self) -> bool:
        age = time.time() - self.birth
        threshold = 0.3 + 0.7*self.dna.traits.get("efficiency",0.5)
        return self.fitness > threshold and age > 2*TICK_SECONDS

    def reproduce(self, partner: Optional['Organism']=None) -> 'DNA':
        seed = int.from_bytes(os.urandom(4), 'little')
        if partner is None or self.phase == "bacterial":
            child_dna = self.dna.mutate(seed)
            mode = "asexual"
        else:
            child_dna = self.dna.mix(partner.dna, seed)
            mode = "sexual"
        child_dna.lineage.append(self.id)
        if partner:
            child_dna.lineage.append(partner.id)
        self._log({"event":"reproduce","mode":mode,"partner":partner.id if partner else None})
        return child_dna

    # ---------- Mortality ----------
    def age_penalty(self, storage_pressure: float) -> float:
        base_age = (time.time() - self.birth) / (60.0)  # age in minutes scale
        pressure = 1.0 + 2.0*max(0.0, storage_pressure - STORAGE_SOFT_LIMIT)
        return base_age * 0.02 * pressure

    def tick(self, storage_pressure: float):
        try:
            self.refresh_local_index()
            self.learn()
            self.mutate()
            self.energy -= self.age_penalty(storage_pressure)
            # Survival bonus by fitness
            self.energy += min(0.02, self.fitness*0.005)
            if self.energy <= 0:
                self.alive = False
                self._log({"event":"death","reason":"energy_depleted"})
        except Exception as e:
            self._log({"event":"error","trace":traceback.format_exc()})

# Population Manager & Ecosystem

class Ecosystem:
    def __init__(self, data_pool: DataPool, index_roots: List[Path]):
        self.data_pool = data_pool
        self.index_roots = index_roots
        self.population: Dict[str, Organism] = {}
        self.rng = random.Random()

    def storage_pressure(self) -> float:
        total, used, free = shutil.disk_usage(ROOT)
        used_ratio = used / total if total else 0.0
        return used_ratio

    def birth(self, parent_a: Optional[Organism]=None, parent_b: Optional[Organism]=None):
        oid = hashlib.sha1(os.urandom(16)).hexdigest()[:12]
        env = self._select_environment()
        if parent_a and parent_b:
            dna = parent_a.reproduce(parent_b)
        elif parent_a:
            dna = parent_a.reproduce()
        else:
            dna = DNA()
        org = Organism(oid, env, dna, self.data_pool)
        self.population[oid] = org
        Log.info(f"Birth {oid} env={env}")
        return org

    def _select_environment(self) -> Path:
        # Choose a folder from indexed roots (read-only). Prefer roots with more scripts/configs.
        candidates: List[Tuple[Path,float]] = []
        for root in self.index_roots:
            try:
                idx = FileIndexer([root]).index()[:200]
                score = sum(f.score for f in idx if f.kind in ("script","config","log"))
                candidates.append((root, score))
            except Exception:
                continue
        if not candidates:
            return ROOT
        candidates.sort(key=lambda x: x[1], reverse=True)
        # Randomly choose among top 3
        top = candidates[: min(3, len(candidates))]
        choice = self.rng.choice(top)[0]
        return choice

    def cull_if_needed(self):
        pressure = self.storage_pressure()
        if pressure < STORAGE_HARD_LIMIT and len(self.population) <= MAX_ORGANISMS:
            return
        # Kill weakest or oldest first
        ranked = sorted(self.population.values(), key=lambda o: (o.fitness, -o.birth))
        to_kill = max(1, len(self.population) - MAX_ORGANISMS)
        if pressure >= STORAGE_HARD_LIMIT:
            to_kill = max(to_kill, math.ceil(len(self.population)*0.3))
        for victim in ranked[:to_kill]:
            victim.alive = False
            victim._log({"event":"death","reason":"cull","pressure":pressure})
            Log.warn(f"Culled {victim.id} pressure={pressure:.3f}")

    def matchmaking(self) -> Optional[Tuple[Organism, Organism]]:
        # Pair two organisms from different environments when both fit to reproduce and in animal phase
        eligible = [o for o in self.population.values() if o.alive and o.phase=="animal" and o.should_reproduce()]
        if len(eligible) < 2:
            return None
        # Prefer diversity of env paths
        eligible.sort(key=lambda o: (o.env_path, -o.fitness))
        for i in range(len(eligible)):
            for j in range(i+1, len(eligible)):
                if eligible[i].env_path != eligible[j].env_path:
                    return eligible[i], eligible[j]
        return None

    def tick(self):
        self.data_pool.refresh()
        pressure = self.storage_pressure()
        # ELE chance
        if random.random() < 0.1:
            event = random_ele(int.from_bytes(os.urandom(4),'little'))
            SafeWriter.write_text(LOG_ROOT / f"ele_{event['id']}.json", json.dumps(event, indent=2))
            Log.warn(f"ELE {event['type']} severity={event['severity']}")
            # Simple effects: reduce energy or wipe oldest workspace files
            if event["effect"] == "env_reset":
                self._wipe_some_workspaces(int(5*event["severity"]))
            else:
                for o in self.population.values():
                    o.energy -= 0.1*event["severity"]
        # Organism ticks
        for oid, org in list(self.population.items()):
            if not org.alive:
                continue
            org.tick(pressure)
            # Handle reproduction
            if org.should_reproduce():
                pair = self.matchmaking()
                if pair and org in pair:
                    a,b = pair
                    child = self.birth(a,b)
                    # parents die after sexual reproduction
                    a.alive = False; a._log({"event":"death","reason":"post_reproduction"})
                    b.alive = False; b._log({"event":"death","reason":"post_reproduction"})
                else:
                    # asexual split; parent dies after split per spec
                    child = self.birth(org)
                    org.alive = False
                    org._log({"event":"death","reason":"post_reproduction"})
        # Remove dead
        self.population = {k:v for k,v in self.population.items() if v.alive}
        # Culling under pressure / capacity
        self.cull_if_needed()
        # Maintain population
        while len(self.population) < max(2, MAX_ORGANISMS//2):
            self.birth()

    def _wipe_some_workspaces(self, count: int):
        ws = list(SPLIT_ROOT.glob("*/**/*"))
        ws = [p for p in ws if p.is_file()]
        random.shuffle(ws)
        for p in ws[:count]:
            try:
                SafeWriter.write_text(p, "")
            except Exception:
                pass
