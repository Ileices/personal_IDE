class ReadOnlySource:
    def __init__(self):
        self.corpus = {}  # Light/electricity stable corpus
        self.neural_maps = {}  # Deposited neural maps
        self.manifests = {}  # Cycle manifests
        self.write_lock = True  # Immutable by default
        self.deposit_queue = []  # Queue for Λ-gated deposits
        
    def attempt_write(self, data: Dict, deposit_type: str) -> bool:
        """
        AE is write-locked except during absularity events
        Returns True if write is allowed (during deposit window)
        """
        if self.write_lock:
            # Queue for later deposit during absularity
            self.deposit_queue.append({
                'data': data,
                'type': deposit_type,
                'timestamp': time.time(),
                'attempt_id': uuid.uuid4()
            })
            return False
        else:
            # Deposit window is open - process immediately
            return self._process_deposit(data, deposit_type)
    
    def open_deposit_window(self, absularity_event: Dict) -> int:
        """
        Open write window during absularity for deposits
        Returns number of deposits processed
        """
        self.write_lock = False
        deposits_processed = 0
        
        # Process queued deposits
        while self.deposit_queue:
            deposit = self.deposit_queue.pop(0)
            if self._process_deposit(deposit['data'], deposit['type']):
                deposits_processed += 1
        
        # Close window after processing
        self.write_lock = True
        return deposits_processed
    
    def _process_deposit(self, data: Dict, deposit_type: str) -> bool:
        """Process a deposit into the read-only source"""
        try:
            if deposit_type == 'neural_map':
                map_id = data['map_id']
                self.neural_maps[map_id] = {
                    'content': data['content'],
                    'rby_pattern': data['rby_pattern'],
                    'compression_schema': data.get('compression_schema'),
                    'deposit_timestamp': time.time(),
                    'absularity_cycle': data.get('cycle_id')
                }
                
            elif deposit_type == 'manifest':
                manifest_id = data['manifest_id']
                self.manifests[manifest_id] = {
                    'content': data['content'],
                    'merkle_root': data['merkle_root'],
                    'cycle_metrics': data.get('cycle_metrics', {}),
                    'deposit_timestamp': time.time()
                }
                
            elif deposit_type == 'excretion':
                # Store excretions with compression metadata
                excretion_id = data['excretion_id']
                self.corpus[excretion_id] = {
                    'content': data['content'],
                    'rby_values': data['rby_values'],
                    'compression_state': 'raw',  # raw, compressed, glyph
                    'access_frequency': 0,
                    'creation_time': time.time(),
                    'last_accessed': time.time()
                }
            
            return True
            
        except Exception as e:
            print(f"Deposit failed: {e}")
            return False
    
    def query_corpus(self, query: Dict) -> List[Dict]:
        """Query the read-only corpus for relevant information"""
        results = []
        query_embedding = query.get('embedding')
        query_rby = query.get('rby_pattern')
        
        for item_id, item in self.corpus.items():
            score = 0.0
            
            # Semantic similarity
            if query_embedding is not None and 'embedding' in item:
                similarity = cosine_similarity(query_embedding, item['embedding'])
                score += similarity * 0.6
            
            # RBY pattern matching
            if query_rby is not None and 'rby_values' in item:
                rby_similarity = 1.0 - np.linalg.norm(query_rby - item['rby_values'])
                score += rby_similarity * 0.4
            
            if score > 0.5:  # Relevance threshold
                results.append({
                    'item_id': item_id,
                    'content': item['content'],
                    'score': score,
                    'compression_state': item['compression_state']
                })
        
        return sorted(results, key=lambda x: x['score'], reverse=True)[:10]

            'parent_id': self.parent_id,

        # Step 3: Validate reconstructability
        reconstructability = self._validate_reconstructability(color_encoded, rarely_used)
        
        if reconstructability >= SystemConstants.MIN_RECONSTRUCTABILITY:
            # Step 4: Apply compression
            compression_result = self._apply_compression(color_encoded, state)
            
            # Step 5: Update storage tracking
            space_freed = self._update_storage_after_compression(compression_result)
            
            compression_event = {

                'space_freed': space_freed,
                'reconstructability': reconstructability,
                'new_usage': self.get_storage_usage()
            }

                current_time = time.time()
                
                # Compute access score (lower = less frequently used)

                access_score = access_frequency / (time_since_access + 1)  # Add 1 to avoid division by zero
                
                excretion_access.append({
                    'excretion_id': excretion_id,
                    'excretion': excretion,
                    'access_score': access_score
                })
        
        # Sort by access score (ascending = rarely used first)
        excretion_access.sort(key=lambda x: x['access_score'])
        
        # Take bottom 20% for compression
        compression_count = max(1, len(excretion_access) // 5)
        return [item['excretion'] for item in excretion_access[:compression_count]]
