            'chunk': data_size * 1.2,

        for granularity in ['binary', 'keystroke', 'token', 'word', 'sentence',

class CoverageMatrix:

    def __init__(self):
        self.interaction_graph = nx.Graph()
        self.coverage_stats = {
            'total_possible_interactions': 0,

            'coverage_ratio': 0.0,
            'unique_patterns': set()
        }
    
    def record_interaction(self, member1_id: str, member2_id: str, 
                         interaction_data: Dict):
        """Record an interaction between two members"""
        # Add edge to interaction graph
        if not self.interaction_graph.has_edge(member1_id, member2_id):
            self.interaction_graph.add_edge(member1_id, member2_id)
            self.coverage_stats['covered_interactions'] += 1
        
        # Record interaction pattern
        pattern_signature = self._compute_pattern_signature(interaction_data)
        self.coverage_stats['unique_patterns'].add(pattern_signature)
        
        # Update coverage ratio
        self._update_coverage_ratio()
    
    def get_coverage_metrics(self) -> Dict:
        """Get current coverage metrics"""
        total_nodes = self.interaction_graph.number_of_nodes()
        total_possible = (total_nodes * (total_nodes - 1)) // 2 if total_nodes > 1 else 0
        
        return {
            'total_members': total_nodes,
            'total_possible_interactions': total_possible,
            'covered_interactions': self.coverage_stats['covered_interactions'],
            'coverage_ratio': (self.coverage_stats['covered_interactions'] / 
                             max(1, total_possible)),
            'unique_patterns_count': len(self.coverage_stats['unique_patterns']),
            'graph_density': nx.density(self.interaction_graph)

    def _compute_composite_volume(self, metrics: Dict) -> float:
        """Compute composite volume from metrics"""
        weights = [0.3, 0.2, 0.2, 0.15, 0.15]  # α1-α5

            metrics.get('artifact_count', 0) / 1000.0,  # Normalized

            metrics.get('evaluation_score', 0),
            metrics.get('compression_ratio', 0)
