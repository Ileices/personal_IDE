class FileSystemScanner:
    """Scan file system and categorize files for IC-AE infection"""
    
    def __init__(self):
        self.file_registry = {}
        self.categorization_engine = FileCategorizationEngine()
        
    def scan_directory(self, root_path: str) -> Dict:
        """Scan directory and return categorized file structure"""
        file_structure = {
            'root_path': root_path,
            'files': [],
            'directories': [],
            'project_roots': [],
            'file_count': 0,
            'total_size': 0
        }
        
        for root, dirs, files in os.walk(root_path):
            # Skip system and identity directories
            if self._is_identity_directory(root):
                continue
                
            # Process files in current directory
            for filename in files:
                file_path = os.path.join(root, filename)
                file_info = self._analyze_file(file_path)
                
                if file_info:
                    file_structure['files'].append(file_info)
                    file_structure['file_count'] += 1
                    file_structure['total_size'] += file_info['size_bytes']
            
            # Check if current directory is a project root
            if self._is_project_root(root, files, dirs):
                project_info = self._analyze_project_root(root, files, dirs)
                file_structure['project_roots'].append(project_info)
        
        return file_structure
    
    def _is_identity_directory(self, path: str) -> bool:
        """Check if directory is an identity folder (Axiom A5)"""
        identity_indicators = ['documents', 'desktop', 'downloads', 'pictures', 
                              'music', 'videos', 'system32', 'program files']
        path_lower = path.lower()
        return any(indicator in path_lower for indicator in identity_indicators)
    
    def _is_project_root(self, directory: str, files: List[str], dirs: List[str]) -> bool:
        """Check if directory qualifies as a project root"""
        # Check for project indicators
        project_files = ['makefile', 'cmakelists.txt', 'package.json', 'setup.py',
                        'pom.xml', 'build.gradle', '.git', 'requirements.txt']
        
        project_dirs = ['src', 'lib', 'include', 'source', 'build', 'dist']
        
        # Count project indicators
        indicator_count = 0
        
        # Check files
        for proj_file in project_files:
            if any(proj_file in f.lower() for f in files):
                indicator_count += 1
        
        # Check directories
        for proj_dir in project_dirs:
            if any(proj_dir in d.lower() for d in dirs):
                indicator_count += 1
        
        return indicator_count >= 2  # At least 2 strong indicators
    
    def _analyze_file(self, file_path: str) -> Dict:
        """Analyze individual file and extract metadata"""
        try:
            stats = os.stat(file_path)
            file_extension = os.path.splitext(file_path)[1].lower()
            
            # Determine file type
            file_type = self.categorization_engine.categorize_file(file_path, file_extension)
            
            # Extract basic content for analysis
            content_preview = self._get_content_preview(file_path, file_type)
            
            file_info = {
                'path': file_path,
                'name': os.path.basename(file_path),
                'size_bytes': stats.st_size,
                'modified_time': stats.st_mtime,
                'file_type': file_type,
                'extension': file_extension,
                'content_preview': content_preview,
                'metadata': self._extract_file_metadata(file_path, file_type)
            }
            
            return file_info
            
        except Exception as e:
            print(f"Error analyzing file {file_path}: {e}")
            return None
    
    def _get_content_preview(self, file_path: str, file_type: str) -> str:
        """Get content preview based on file type"""
        try:
            if file_type == 'text' or file_type == 'code':
                with open(file_path, 'r', encoding='utf-8', errors='ignore') as f:
                    return f.read(1000)  # First 1000 characters
            elif file_type == 'data':
                with open(file_path, 'r', encoding='utf-8', errors='ignore') as f:
                    return f.read(500)  # First 500 characters for data files
            else:
                return ""  # Binary files get no content preview
        except:
            return ""

class CoverageTracker:
    """Track infection coverage and interaction patterns"""

        """Record interaction between two files"""
        # Add to interaction graph
        if not self.interaction_graph.has_edge(file1, file2):
            self.interaction_graph.add_edge(file1, file2, **interaction_data)
            self.coverage_metrics['realized_interactions'] += 1

        pattern_signature = self._compute_pattern_signature(interaction_data)
        self.coverage_metrics['unique_patterns'].add(pattern_signature)

        """Record propagation results"""
        self.coverage_metrics['infected_files'].update(
            propagation_result.get('newly_infected', [])

        file_coverage = len(self.coverage_metrics['infected_files']) / total_files
        
        total_possible = self.coverage_metrics['total_possible_interactions']
        if total_possible > 0:
            interaction_coverage = self.coverage_metrics['realized_interactions'] / total_possible
        else:
            interaction_coverage = 0.0
        
        # Weighted combination
        return 0.6 * file_coverage + 0.4 * interaction_coverage

            'interaction_count': self.coverage_metrics['realized_interactions'],

            'understanding_level': self.system_state['understanding_level'],
            'propagation_depth': self.system_state['infection_depth'],
            'neural_data_generated': len(neural_data),
            'file_structure': file_structure
        }
        
        self.operation_log.append(cycle_result)
        return cycle_result
    
    def _create_initial_infections(self, file_structure: Dict) -> List[Dict]:
        """Create initial IC-AE infections from file structure"""
        initial_infections = []
        
        # Process project roots first
        for project in file_structure.get('project_roots', []):
            project_files = self._get_project_files(project, file_structure)
            if project_files:
                sandbox = self.sandbox_manager.create_sandbox(project_files, 0)
                initial_infections.append(sandbox)
        
        # Process standalone files
        standalone_files = self._get_standalone_files(file_structure)
        for file_group in self._group_files_by_type(standalone_files):
            if file_group:
                sandbox = self.sandbox_manager.create_sandbox(file_group, 0)
                initial_infections.append(sandbox)
        
        return initial_infections
    
    def _propagate_infections_hierarchically(self, initial_infections: List[Dict]) -> List[Dict]:
        """Propagate infections hierarchically through depth levels"""
        all_propagation_results = []
        current_sandboxes = initial_infections.copy()
        max_depth = self.config.get('max_infection_depth', 3)
        
        for depth in range(1, max_depth + 1):
            depth_propagation_results = []
            
            for sandbox in current_sandboxes:
                # Find files that can be infected from this sandbox
                target_files = self._find_infection_targets(sandbox, depth)
                
                if target_files:
                    propagation_result = self.infection_propagator.propagate_infection(
                        sandbox, target_files, depth
                    )
                    depth_propagation_results.append(propagation_result)
                    
                    # Add new sandboxes to current list for next depth
                    new_sandbox_ids = propagation_result.get('new_sandboxes_created', [])
                    for sandbox_id in new_sandbox_ids:
                        if sandbox_id in self.sandbox_manager.active_sandboxes:
                            current_sandboxes.append(
                                self.sandbox_manager.active_sandboxes[sandbox_id]
                            )
            
            all_propagation_results.extend(depth_propagation_results)
            
            # Stop if coverage is sufficient or no new infections
            if (self.system_state['current_coverage'] > 0.95 or 
                not depth_propagation_results):
                break
        
        self.system_state['infection_depth'] = depth
        return all_propagation_results
    
    def _generate_neural_compression_data(self, propagation_results: List[Dict]) -> List[Dict]:
        """Generate neural compression data from infection results"""
        neural_data = []
        
        for result in propagation_results:
            # Extract patterns for neural compression
            patterns = self._extract_compression_patterns(result)
            
            # Generate absoleices (memory units)
            absoleices = self._generate_absoleices(patterns, result)
            
            # Create neural maps
            neural_map = self._create_neural_map(patterns, absoleices)
            
            neural_data.append({
                'propagation_result_id': result.get('source_sandbox', 'unknown'),
                'patterns': patterns,
                'absoleices': absoleices,
                'neural_map': neural_map,
                'compression_ratio': self._compute_compression_ratio(patterns),
                'generation_time': time.time()
            })
        
        return neural_data
    
    def _extract_compression_patterns(self, propagation_result: Dict) -> List[Dict]:
        """Extract patterns suitable for neural compression"""
        patterns = []
        
        # Extract file interaction patterns
        interactions = propagation_result.get('interactions_generated', [])
        for interaction in interactions:
            pattern = {
                'type': 'file_interaction',
                'files': interaction.get('files', []),
                'rby_pattern': interaction.get('rby_signature', []),
                'interaction_strength': interaction.get('strength', 0.5),
                'compression_potential': interaction.get('compression_potential', 0.5)
            }
            patterns.append(pattern)
        
        # Extract execution patterns
        execution_data = propagation_result.get('execution_results', {})
        if execution_data:
            pattern = {
                'type': 'execution_pattern',
                'language': execution_data.get('language', 'unknown'),
                'success_rate': execution_data.get('success', False),
                'execution_time': execution_data.get('execution_time', 0),
                'common_errors': execution_data.get('errors', '')
            }
            patterns.append(pattern)
        
        return patterns
